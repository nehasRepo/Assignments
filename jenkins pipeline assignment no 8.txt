
pipeline {

    agent {
        label {
            label 'Master'
            customWorkspace '/mnt/repo1'
        }
    }

    stages {

        stage('Cloning Repo1') {
            steps {
                git branch: 'main',
                    url: 'https://github.com/nehasRepo/jenkins.git'
            }
        }

        stage('Deploying Repo1 on Master') {
            steps {
                sh '''
                    sudo yum install httpd -y
                    sudo systemctl start httpd
                    sudo systemctl enable httpd
                    sudo cp index.html /var/www/html/index.html
                    sudo chmod 644 /var/www/html/index.html
                '''
            }
        }

        stage('Cloning Repo2') {

            agent {
                label {
                    label 'slave1'
                    customWorkspace '/mnt/repo2'
                }
            }

            steps {
                git branch: 'main',
                    url: 'https://github.com/nehasRepo/jenkins2.git'
            }
        }

        stage('Deploying Repo2 on Slave1') {

            agent {
                label {
                    label 'slave1'
                    customWorkspace '/mnt/repo2'
                }
            }

            steps {
                sh '''
                    sudo yum install httpd -y
                    sudo systemctl start httpd
                    sudo systemctl enable httpd
                    sudo cp index.html /var/www/html/index.html
                    sudo chmod 644 /var/www/html/index.html
                '''
            }
        }

        stage('Cloning Repo3') {

            agent {
                label {
                    label 'slave2'
                    customWorkspace '/mnt/repo3'
                }
            }

            steps {
                git branch: 'main',
                    url: 'https://github.com/nehasRepo/jenkins-3.git'
            }
        }

        stage('Deploying Repo3 on Slave2') {

            agent {

                    customWorkspace '/mnt/repo3'

            steps {
                sh '''
                    sudo yum install httpd -y
                    sudo systemctl start httpd
                    sudo systemctl enable httpd
                    sudo cp index.html /var/www/html/index.html
                    sudo chmod 644 /var/www/html/index.html
                '''
            }
        }
    }
