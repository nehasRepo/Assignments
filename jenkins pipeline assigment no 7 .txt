            }
        }

        stage('Installing HTTPD') {
            steps {
                sh 'sudo yum install httpd -y'
                sh 'sudo systemctl enable httpd'
            }
        }

        stage('Deploy Repo 1') {
            steps {
                sh 'sudo cp repo1/index.html /var/www/html/index.html'
