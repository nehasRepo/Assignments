pipeline {
    agent any

    stages {
        stage('Deploying Application') {
            steps {
                sh 'sudo yum install httpd -y'
                sh 'sudo cp index.html /var/www/html/'
                sh 'sudo chmod 644 /var/www/html/index.html'
                sh 'sudo systemctl restart httpd'
            }
        }
    }
}
